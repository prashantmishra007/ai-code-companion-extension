/**
 * @file CppTemplateGenerator.js
 * @description A production-ready utility to generate, customize, and save C++ starter templates.
 */

const fs = require('fs').promises;
const path = require('path');

class CppTemplateGenerator {
  /**
   * Map of available C++ templates
   */
  static TEMPLATES = {
    helloWorld: `/**
 * Program: Hello World
 * Author: {{AUTHOR}}
 * Date: {{DATE}}
 */
#include <iostream>

int main() {
    std::cout << "Hello, World!" << std::endl;
    return 0;
}
`,

    competitiveProgramming: `/**
 * Program: Competitive Programming Starter
 * Author: {{AUTHOR}}
 * Date: {{DATE}}
 */
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>

using namespace std;

void solve() {
    // Add your logic here
}

int main() {
    // Optimize standard I/O operations
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int t = 1;
    // cin >> t; // Uncomment if test cases are provided
    while (t--) {
        solve();
    }

    return 0;
}
`,

    oopExample: `/**
 * Program: Object-Oriented Programming Template
 * Author: {{AUTHOR}}
 * Date: {{DATE}}
 */
#include <iostream>
#include <string>

class {{CLASS_NAME}} {
private:
    std::string name;
    int id;

public:
    {{CLASS_NAME}}(std::string objName, int objId) : name(objName), id(objId) {}

    void displayInfo() const {
        std::cout << "Name: " << name << ", ID: " << id << std::endl;
    }
};

int main() {
    {{CLASS_NAME}} obj("SampleObject", 101);
    obj.displayInfo();
    return 0;
}
`
  };

  /**
   * Helper method to replace template variables/placeholders.
   * @private
   * @param {string} templateStr 
   * @param {Object.<string, string>} variables 
   * @returns {string}
   */
  static _render(templateStr, variables = {}) {
    const defaults = {
      AUTHOR: 'Developer',
      DATE: new Date().toISOString().split('T')[0],
      CLASS_NAME: 'Person',
      ...variables
    };

    return Object.entries(defaults).reduce((acc, [key, value]) => {
      const regex = new RegExp(`\\{\\{${key}\\}\\}`, 'g');
      return acc.replace(regex, value);
    }, templateStr);
  }

  /**
   * Register a custom C++ template dynamically.
   * @param {string} name - Unique identifier for the template
   * @param {string} templateContent - Multi-line C++ template string
   */
  static registerTemplate(name, templateContent) {
    if (!name || typeof name !== 'string') {
      throw new Error('Template name must be a valid string.');
    }
    if (!templateContent || typeof templateContent !== 'string') {
      throw new Error('Template content must be a valid string.');
    }
    this.TEMPLATES[name] = templateContent;
  }

  /**
   * Get a C++ template by type with dynamic variables rendered.
   * @param {string} [type='helloWorld']
   * @param {Object.<string, string>} [variables={}]
   * @returns {string}
   */
  static getTemplate(type = 'helloWorld', variables = {}) {
    const template = this.TEMPLATES[type];
    if (!template) {
      throw new Error(
        `Template '${type}' not found. Available templates: ${Object.keys(this.TEMPLATES).join(', ')}`
      );
    }
    return this._render(template, variables);
  }

  /**
   * Saves the generated C++ code to a specified file.
   * @param {string} destinationPath - Path where the file should be saved.
   * @param {string} [templateType='helloWorld']
   * @param {Object} [options={}]
   * @param {Object.<string, string>} [options.variables={}] - Placeholders replacement map
   * @param {boolean} [options.overwrite=true] - Whether to overwrite if the file exists
   * @returns {Promise<string>} Absolute path to the created file.
   */
  static async saveToFile(destinationPath, templateType = 'helloWorld', options = {}) {
    const { variables = {}, overwrite = true } = options;

    if (!destinationPath || typeof destinationPath !== 'string') {
      throw new Error('A valid destination path is required.');
    }

    const code = this.getTemplate(templateType, variables);
    const resolvedPath = path.resolve(destinationPath);

    if (!overwrite) {
      try {
        await fs.access(resolvedPath);
        throw new Error(`File already exists at '${resolvedPath}' and overwrite is set to false.`);
      } catch (error) {
        if (error.code !== 'ENOENT') throw error; // Rethrow if it's not a "file doesn't exist" error
      }
    }

    // Ensure directory exists
    await fs.mkdir(path.dirname(resolvedPath), { recursive: true });
    await fs.writeFile(resolvedPath, code, 'utf-8');

    return resolvedPath;
  }
}

// Usage Example

  try {

// function to find factorial of a number
function factorial(n) {
  if (n < 0) return -1;
  if (n === 0 || n === 1) return 1;
  return n * factorial(n - 1);
}
