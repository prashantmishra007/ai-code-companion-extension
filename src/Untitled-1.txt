// Function to calculate user discount and format currency
function applyDiscount(price, discountPercent) {
    if (typeof price !== "number" || typeof discountPercent !== "number" || price < 0 || discountPercent < 0 || discountPercent > 100) {
        return "Invalid input";
    }
    
    const finalPrice = price * (1 - discountPercent / 100);
    return `$${finalPrice.toFixed(2)}`;
}

const bcrypt = require('bcrypt');

const router = express.Router();

// Temporary in-memory database for demonstration purposes
const users = [];

/**
 * POST /api/register
 * Registers a new user with a hashed password.
 */
router.post('/register', async (req, res) => {
  try {
    const { username, password } = req.body;

    // Validate input
    if (!username || !password) {
      return res.status(400).json({ error: 'Username and password are required' });
    }

    // Check if user already exists
    const existingUser = users.find(u => u.username === username);
    if (existingUser) {
      return res.status(409).json({ error: 'Username already taken' });
    }

    // Define salt rounds for bcrypt
    const saltRounds = 10;

    // Hash the password securely
    const hashedPassword = await bcrypt.hash(password, saltRounds);

    // Save the user (in a real app, this would be a database call)
    const newUser = { username, password: hashedPassword };
    users.push(newUser);

    // Respond with success (excluding the password)
    return res.status(201).json({
      message: 'User registered successfully',
      user: { username }
    });
  } catch (error) {
    console.error('Registration error:', error);
    return res.status(500).json({ error: 'Internal server error' });
  }
});

module.exports = router;
