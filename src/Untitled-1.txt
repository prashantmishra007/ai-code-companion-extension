/**
 * Removes duplicate elements from an array and returns a sorted array.
 * 
 * @param {Array} arr - The input array (e.g., numbers or strings)
 * @returns {Array} - A new array with unique elements sorted in ascending order
 */
function removeDuplicatesAndSort(arr) {
  // Validate input
  if (!Array.isArray(arr)) {
    throw new TypeError('Input must be a valid array');
  }

  // Step 1: Remove duplicates using ES6 Set
  const uniqueArray = Array.from(new Set(arr));

  // Step 2: Sort the array
  // Using a custom comparator to correctly sort both numbers and strings
  return uniqueArray.sort((a, b) => {
    if (typeof a === 'number' && typeof b === 'number') {
      return a - b; // Numeric ascending sort
    }
    return String(a).localeCompare(String(b)); // String alphabetical sort
  });
}
